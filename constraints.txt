#   python3 -m venv /tmp/lockenv && /tmp/lockenv/bin/pip install -e ".[test]"
#   /tmp/lockenv/bin/pip freeze --exclude-editable | sort > constraints.txt
#   然後在該 venv 跑一次 `pytest tests/` 確認全綠再提交。
